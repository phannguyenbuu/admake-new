require 'sketchup.rb'

module HealthWaveBillboard3D
  extend self

  MODEL = Sketchup.active_model

  # ----------------------------------------------------------
  # 01. THONG SO CHINH
  # ----------------------------------------------------------

  SIGN_W = 4.80
  SIGN_H = 3.00
  SIGN_T = 0.10
  SIGN_Z = 8.50

  POST_HEIGHT = 8.50
  POST_RADIUS = 0.18
  POST_BOX_W = 0.25

  FOUNDATION_W = 1.20
  FOUNDATION_D = 1.20
  FOUNDATION_H = 0.65

  FRAME_SIZE = 0.06
  FRAME_DEPTH = 0.18

  CATWALK_W = 0.65
  CATWALK_D = 0.70

  BOLT_RADIUS = 0.075
  BOLT_DEPTH = 0.055

  # Mat truoc pano quay ve phia -Y
  PANEL_FRONT_Y = -0.28

  # ----------------------------------------------------------
  # 02. DON VI VA HINH HOC CO BAN
  # ----------------------------------------------------------

  def mm(value)
    value.to_f * 39.37007874015748
  end

  def pt(x, y, z)
    Geom::Point3d.new(mm(x), mm(y), mm(z))
  end

  def vec(x, y, z)
    Geom::Vector3d.new(x, y, z)
  end

  def group(parent, name)
    g = parent.add_group
    g.name = name
    g
  end

  def material(name, rgb, alpha = 1.0)
    mats = MODEL.materials
    mat = mats[name] || mats.add(name)
    mat.color = Sketchup::Color.new(*rgb)
    mat.alpha = alpha
    mat
  end

  def setup_materials
    @concrete = material('HW Concrete', [170, 169, 161])
    @concrete_dark = material('HW Dark Concrete', [115, 117, 115])

    @steel = material('HW Steel', [130, 142, 150])
    @steel_dark = material('HW Dark Steel', [60, 68, 73])
    @stainless = material('HW Stainless Steel', [205, 211, 216])
    @chrome = material('HW Chrome', [225, 228, 230])

    @panel_white = material('HW White Acrylic Panel', [250, 250, 248])
    @panel_edge = material('HW Panel Edge', [160, 171, 178])

    @green_dark = material('HW Logo Dark Green', [0, 75, 48])
    @green = material('HW Logo Green', [90, 160, 45])
    @green_light = material('HW Logo Light Green', [165, 205, 65])
    @orange = material('HW Logo Orange', [238, 112, 24])
    @orange_light = material('HW Logo Light Orange', [255, 177, 28])

    @text_dark = material('HW Text Dark', [0, 37, 29])
    @ground = material('HW Ground', [205, 205, 202])
  end

  # ----------------------------------------------------------
  # 03. KHOI HOP
  # ----------------------------------------------------------

  def box(entities, x, y, z, dx, dy, dz, mat)
    return nil if dx <= 0 || dy <= 0 || dz <= 0

    corners = [
      pt(x,      y,      z),
      pt(x + dx, y,      z),
      pt(x + dx, y + dy, z),
      pt(x,      y + dy, z)
    ]

    face = entities.add_face(corners)
    return nil unless face && face.valid?

    face.reverse! if face.normal.z < 0
    face.material = mat
    face.back_material = mat
    face.pushpull(mm(dz))
    face
  end

  # ----------------------------------------------------------
  # 04. THANH HOP NOI HAI DIEM
  # ----------------------------------------------------------

  def beam(entities, p1, p2, size, mat)
    direction = p2 - p1
    length = direction.length
    return nil if length < 0.001

    direction.normalize!

    reference = vec(0, 0, 1)
    reference = vec(0, 1, 0) if direction.parallel?(reference)

    axis_x = direction.cross(reference)
    axis_x.normalize!

    axis_y = direction.cross(axis_x)
    axis_y.normalize!

    r = mm(size) / 2.0

    corners = [
      p1.offset(axis_x, r).offset(axis_y, r),
      p1.offset(axis_x, -r).offset(axis_y, r),
      p1.offset(axis_x, -r).offset(axis_y, -r),
      p1.offset(axis_x, r).offset(axis_y, -r)
    ]

    face = entities.add_face(corners)
    return nil unless face && face.valid?

    face.material = mat
    face.back_material = mat
    face.pushpull(length)
    face
  end

  # ----------------------------------------------------------
  # 05. ONG TRON DUNG
  # ----------------------------------------------------------

  def cylinder(entities, x, y, z, radius, height, mat, segments = 32)
    edges = entities.add_circle(
      pt(x, y, z),
      vec(0, 0, 1),
      mm(radius),
      segments
    )

    face = entities.add_face(edges)
    return nil unless face && face.valid?

    face.material = mat
    face.back_material = mat
    face.pushpull(mm(height))
    face
  end

  # ----------------------------------------------------------
  # 06. TRU TRON THEP
  # ----------------------------------------------------------

  def horizontal_cylinder(entities, center, radius, length, direction, mat)
    edges = entities.add_circle(
      center,
      direction,
      mm(radius),
      32
    )

    face = entities.add_face(edges)
    return nil unless face && face.valid?

    face.material = mat
    face.back_material = mat
    face.pushpull(mm(length))
    face
  end

  # ----------------------------------------------------------
  # 07. MONG BE TONG VA BAN DE
  # ----------------------------------------------------------

  def build_foundation(parent)
    g = group(parent, '01 - Mong be tong va ban de')
    e = g.entities

    box(
      e,
      -FOUNDATION_W / 2.0,
      -FOUNDATION_D / 2.0,
      -FOUNDATION_H,
      FOUNDATION_W,
      FOUNDATION_D,
      FOUNDATION_H,
      @concrete
    )

    # Be tong lot mong
    box(
      e,
      -FOUNDATION_W / 2.0 - 0.15,
      -FOUNDATION_D / 2.0 - 0.15,
      -FOUNDATION_H - 0.12,
      FOUNDATION_W + 0.30,
      FOUNDATION_D + 0.30,
      0.12,
      @concrete_dark
    )

    # Be co cot
    box(
      e,
      -0.35, -0.35, 0,
      0.70, 0.70, 0.25,
      @concrete
    )

    # Ban de cot thep
    box(
      e,
      -0.34, -0.34, 0.25,
      0.68, 0.68, 0.045,
      @steel_dark
    )

    # Bu long neo va dai oc
    [-0.24, 0.24].each do |x|
      [-0.24, 0.24].each do |y|
        cylinder(e, x, y, 0.18, 0.025, 0.20, @steel_dark, 12)
        cylinder(e, x, y, 0.43, 0.045, 0.025, @chrome, 12)
      end
    end

    g
  end

  # ----------------------------------------------------------
  # 08. COT THEP TRU TRON
  # ----------------------------------------------------------

  def build_round_post(parent)
    g = group(parent, '02 - Cot thep tron')
    e = g.entities

    cylinder(
      e,
      0, 0, 0.295,
      POST_RADIUS,
      POST_HEIGHT,
      @steel,
      48
    )

    # Vong tang cung
    [0.45, 1.0, 4.0, 7.8].each do |z|
      cylinder(
        e,
        0, 0, z,
        POST_RADIUS + 0.018,
        0.045,
        @steel_dark,
        48
      )
    end

    g
  end

  # ----------------------------------------------------------
  # 09. COT THEP HOP PHU
  # ----------------------------------------------------------

  def build_box_post(parent)
    g = group(parent, '03 - Cot thep hop phu')
    e = g.entities

    x = 0.55
    y = 0.15
    w = POST_BOX_W

    box(
      e,
      x - w / 2.0,
      y - w / 2.0,
      0.295,
      w, w,
      POST_HEIGHT - 0.10,
      @steel_dark
    )

    # Nap dau cot
    box(
      e,
      x - w / 2.0 - 0.035,
      y - w / 2.0 - 0.035,
      POST_HEIGHT + 0.16,
      w + 0.07,
      w + 0.07,
      0.06,
      @stainless
    )

    g
  end

  # ----------------------------------------------------------
  # 10. KHUNG XUONG THEP SAU PANO
  # ----------------------------------------------------------

  def build_frame(parent)
    g = group(parent, '04 - Khung xuong thep pano')
    e = g.entities

    x1 = -SIGN_W / 2.0
    x2 = SIGN_W / 2.0
    z1 = SIGN_Z
    z2 = SIGN_Z + SIGN_H

    yf = PANEL_FRONT_Y + SIGN_T
    yb = yf + FRAME_DEPTH

    # Khung chu nhat
    beam(e, pt(x1, yb, z1), pt(x2, yb, z1), FRAME_SIZE, @steel)
    beam(e, pt(x1, yb, z2), pt(x2, yb, z2), FRAME_SIZE, @steel)

    beam(e, pt(x1, yb, z1), pt(x1, yb, z2), FRAME_SIZE, @steel)
    beam(e, pt(x2, yb, z1), pt(x2, yb, z2), FRAME_SIZE, @steel)

    # Thanh dung gia cuong
    [-1.6, -0.8, 0, 0.8, 1.6].each do |x|
      beam(
        e,
        pt(x, yb, z1 + 0.10),
        pt(x, yb, z2 - 0.10),
        0.035,
        @steel_dark
      )
    end

    # Thanh ngang gia cuong
    [z1 + 0.50, z1 + 1.50, z1 + 2.50].each do |z|
      beam(
        e,
        pt(x1 + 0.10, yb, z),
        pt(x2 - 0.10, yb, z),
        0.035,
        @steel
      )
    end

    # Giang cheo
    4.times do |i|
      xa = x1 + i * SIGN_W / 4.0
      xb = xa + SIGN_W / 4.0

      beam(
        e,
        pt(xa, yb, z1),
        pt(xb, yb, z2),
        0.025,
        @steel_dark
      )

      beam(
        e,
        pt(xa, yb, z2),
        pt(xb, yb, z1),
        0.025,
        @steel_dark
      )
    end

    g
  end

  # ----------------------------------------------------------
  # 11. MAT PANO TRANG VA VIEN INOX
  # ----------------------------------------------------------

  def build_panel(parent)
    g = group(parent, '05 - Pano trang va vien inox')
    e = g.entities

    # Tam pano
    box(
      e,
      -SIGN_W / 2.0,
      PANEL_FRONT_Y,
      SIGN_Z,
      SIGN_W,
      SIGN_T,
      SIGN_H,
      @panel_white
    )

    edge = 0.035
    y = PANEL_FRONT_Y - 0.035

    # Vien tren va duoi
    box(
      e,
      -SIGN_W / 2.0,
      y,
      SIGN_Z + SIGN_H - edge,
      SIGN_W,
      0.035,
      edge,
      @panel_edge
    )

    box(
      e,
      -SIGN_W / 2.0,
      y,
      SIGN_Z,
      SIGN_W,
      0.035,
      edge,
      @panel_edge
    )

    # Vien trai va phai
    box(
      e,
      -SIGN_W / 2.0,
      y,
      SIGN_Z,
      edge,
      0.035,
      SIGN_H,
      @panel_edge
    )

    box(
      e,
      SIGN_W / 2.0 - edge,
      y,
      SIGN_Z,
      edge,
      0.035,
      SIGN_H,
      @panel_edge
    )

    g
  end

  # ----------------------------------------------------------
  # 12. NUT BAT INOX TRANG TRI
  # ----------------------------------------------------------

  def build_standoffs(parent)
    g = group(parent, '06 - Nut bat inox')
    e = g.entities

    positions = [
      [-SIGN_W / 2.0 + 0.25, SIGN_Z + SIGN_H - 0.25],
      [ SIGN_W / 2.0 - 0.25, SIGN_Z + SIGN_H - 0.25],
      [-SIGN_W / 2.0 + 0.25, SIGN_Z + 0.25],
      [ SIGN_W / 2.0 - 0.25, SIGN_Z + 0.25]
    ]

    positions.each do |x, z|
      # Chan cach tuong
      horizontal_cylinder(
        e,
        pt(x, PANEL_FRONT_Y - 0.08, z),
        0.055,
        0.12,
        vec(0, 1, 0),
        @steel_dark
      )

      # Dau nut inox
      horizontal_cylinder(
        e,
        pt(x, PANEL_FRONT_Y - 0.18, z),
        BOLT_RADIUS,
        BOLT_DEPTH,
        vec(0, 1, 0),
        @chrome
      )

      # Mat dau nut
      horizontal_cylinder(
        e,
        pt(x, PANEL_FRONT_Y - 0.235, z),
        BOLT_RADIUS * 0.72,
        0.008,
        vec(0, 1, 0),
        @stainless
      )
    end

    g
  end

  # ----------------------------------------------------------
  # 13. HINH HOC LOGO
  # ----------------------------------------------------------

  def add_disc(entities, x, y, z, radius, depth, mat)
    horizontal_cylinder(
      entities,
      pt(x, y, z),
      radius,
      depth,
      vec(0, 1, 0),
      mat
    )
  end

  def add_flat_polygon(entities, points, y, mat, depth = 0.018)
    pts = points.map { |x, z| pt(x, y, z) }
    face = entities.add_face(pts)
    return nil unless face && face.valid?

    face.material = mat
    face.back_material = mat

    # Day vat lieu ra phia truoc pano
    face.pushpull(mm(depth))
    face
  end

  def add_arc_band(entities, cx, cz, outer_r, inner_r,
                   start_angle, end_angle, mat, y, segments = 36)
    points = []

    segments.times do |i|
      a = start_angle + (end_angle - start_angle) * i / (segments - 1).to_f
      points << [cx + outer_r * Math.cos(a), cz + outer_r * Math.sin(a)]
    end

    segments.times do |i|
      a = end_angle - (end_angle - start_angle) * i / (segments - 1).to_f
      points << [cx + inner_r * Math.cos(a), cz + inner_r * Math.sin(a)]
    end

    add_flat_polygon(entities, points, y, mat, 0.025)
  end

  def build_logo(parent)
    g = group(parent, '07 - Logo HealthWave')
    e = g.entities

    cx = 0.0
    cz = SIGN_Z + 2.05
    y = PANEL_FRONT_Y - 0.045

    # Vong cung cam lon
    add_arc_band(
      e, cx, cz, 0.79, 0.68,
      0.55, 5.15,
      @orange, y
    )

    # Vong cung xanh la ben phai
    add_arc_band(
      e, cx + 0.04, cz + 0.02,
      0.62, 0.53,
      -1.10, 1.25,
      @green, y - 0.012
    )

    # Vong cung xanh dam phia trong
    add_arc_band(
      e, cx + 0.02, cz + 0.01,
      0.49, 0.42,
      -1.20, 1.65,
      @green_dark, y - 0.020
    )

    # La trai
    add_flat_polygon(
      e,
      [
        [-0.40, cz + 0.30],
        [-0.17, cz + 0.20],
        [-0.04, cz - 0.05],
        [-0.12, cz - 0.32],
        [-0.28, cz - 0.08]
      ],
      y - 0.04,
      @green,
      0.035
    )

    # La phai
    add_flat_polygon(
      e,
      [
        [0.02, cz - 0.31],
        [0.22, cz - 0.17],
        [0.43, cz - 0.23],
        [0.25, cz - 0.38],
        [0.08, cz - 0.39]
      ],
      y - 0.05,
      @green_dark,
      0.035
    )

    # Than cay uon
    beam(
      e,
      pt(-0.02, y - 0.06, cz - 0.40),
      pt(0.05, y - 0.06, cz - 0.10),
      0.025,
      @green_dark
    )

    # Dau hinh tron mau cam
    add_disc(
      e,
      0.20, y - 0.07, cz - 0.02,
      0.075, 0.025, @orange
    )

    g
  end

  # ----------------------------------------------------------
  # 14. CHU NOI 3D
  # ----------------------------------------------------------

  def add_3d_text(entities, text, point, height, mat,
                  alignment = TextAlignCenter)
    text_group = entities.add_group
    text_group.name = "Text - #{text}"

    text_entity = text_group.entities.add_3d_text(
      text,
      alignment,
      'Arial',
      false,
      true,
      mm(height),
      0.0,
      0.0,
      true,
      mm(height * 0.10)
    )

    text_entity.material = mat if text_entity.respond_to?(:material=)

    # Chu SketchUp tao tren mat phang XY, xoay de mat chu huong ra -Y
    transform = Geom::Transformation.new(point)
    rotate = Geom::Transformation.rotation(
      ORIGIN,
      X_AXIS,
      90.degrees
    )

    text_group.transform!(transform * rotate)
    text_group
  end

  def build_text(parent)
    g = group(parent, '08 - Chu noi HealthWave')
    e = g.entities

    # Chu chinh
    add_3d_text(
      e,
      'HealthWave',
      pt(0.0, PANEL_FRONT_Y - 0.10, SIGN_Z + 0.72),
      0.46,
      @text_dark,
      TextAlignCenter
    )

    # Dong tagline
    add_3d_text(
      e,
      'YOUR BUSINESS TAGLINE GOES HERE',
      pt(0.0, PANEL_FRONT_Y - 0.10, SIGN_Z + 0.42),
      0.135,
      @text_dark,
      TextAlignCenter
    )

    g
  end

  # ----------------------------------------------------------
  # 15. SAN CATWALK VA LAN CAN
  # ----------------------------------------------------------

  def build_catwalk(parent)
    g = group(parent, '09 - San catwalk va lan can')
    e = g.entities

    z = SIGN_Z - 0.15
    x1 = -SIGN_W / 2.0 - 0.15
    x2 = SIGN_W / 2.0 + 0.15
    y1 = 0.15
    y2 = y1 + CATWALK_D

    # Mat san
    box(
      e,
      x1, y1, z,
      x2 - x1, CATWALK_D, 0.06,
      @steel_dark
    )

    # Dam doc
    [y1, y2].each do |y|
      beam(
        e,
        pt(x1, y, z - 0.05),
        pt(x2, y, z - 0.05),
        0.05, @steel
      )
    end

    # Lan can
    rail_z = z + 1.0

    [y1, y2].each do |y|
      beam(
        e,
        pt(x1, y, rail_z),
        pt(x2, y, rail_z),
        0.035, @steel
      )

      beam(
        e,
        pt(x1, y, z + 0.50),
        pt(x2, y, z + 0.50),
        0.025, @steel
      )

      12.times do |i|
        x = x1 + (x2 - x1) * i / 11.0

        beam(
          e,
          pt(x, y, z),
          pt(x, y, rail_z),
          0.025, @steel
        )
      end
    end

    g
  end

  # ----------------------------------------------------------
  # 16. THANG LEO AN TOAN
  # ----------------------------------------------------------

  def build_ladder(parent)
    g = group(parent, '10 - Thang leo an toan')
    e = g.entities

    x = 0.45
    y = 0.25
    z0 = 0.40
    z1 = SIGN_Z - 0.05

    # Thanh doc
    [-0.18, 0.18].each do |dx|
      beam(
        e,
        pt(x + dx, y, z0),
        pt(x + dx, y, z1),
        0.035, @steel
      )
    end

    # Bac thang
    z = z0

    while z < z1
      beam(
        e,
        pt(x - 0.18, y, z),
        pt(x + 0.18, y, z),
        0.025, @steel_light
      )
      z += 0.28
    end

    # Long bao ve phia tren
    cage_y = y + 0.20
    cage_start = 3.0

    [-0.35, 0.35].each do |dx|
      beam(
        e,
        pt(x + dx, cage_y, cage_start),
        pt(x + dx, cage_y, z1),
        0.025, @steel
      )
    end

    z = cage_start

    while z < z1
      beam(
        e,
        pt(x - 0.35, cage_y, z),
        pt(x + 0.35, cage_y, z),
        0.022, @steel
      )
      z += 0.55
    end

    g
  end

  # ----------------------------------------------------------
  # 17. MAT DAT THAM KHAO
  # ----------------------------------------------------------

  def build_ground(parent)
    g = group(parent, '00 - Mat dat tham khao')

    box(
      g.entities,
      -8, -6, -0.90,
      16, 12, 0.20,
      @ground
    )

    g
  end

  # ----------------------------------------------------------
  # 18. CAMERA
  # ----------------------------------------------------------

  def setup_camera
    view = MODEL.active_view

    camera = Sketchup::Camera.new(
      pt(7.5, -12.0, 11.0),
      pt(0.0, 0.0, 6.5),
      vec(0, 0, 1)
    )

    view.camera = camera
    view.zoom_extents
    view.invalidate
  end

  # ----------------------------------------------------------
  # 19. TAO TOAN BO MO HINH
  # ----------------------------------------------------------

  def build
    MODEL.start_operation('Create HealthWave Billboard', true)

    begin
      setup_materials

      root = group(
        MODEL.active_entities,
        'HEALTHWAVE - COMPLETE SIGN MODEL'
      )

      build_ground(root)
      build_foundation(root)
      build_round_post(root)
      build_box_post(root)
      build_frame(root)
      build_panel(root)
      build_standoffs(root)
      build_logo(root)
      build_text(root)
      build_catwalk(root)
      build_ladder(root)

      MODEL.commit_operation
      setup_camera

      puts 'HealthWave Billboard 3D created successfully.'

    rescue StandardError => error
      MODEL.abort_operation
      puts "Error: #{error.message}"
      puts error.backtrace
    end
  end
end

HealthWaveBillboard3D.build
