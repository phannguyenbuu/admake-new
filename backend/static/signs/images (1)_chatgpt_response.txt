require 'sketchup.rb'

module SignBoardMockup
  extend self

  # --------------------------------------------------------------
  # CAU HINH
  # --------------------------------------------------------------

  BOARD_W = 8.4
  BOARD_H = 2.65
  BOARD_D = 0.32

  BOARD_X = 0.0
  BOARD_Y = -0.34
  BOARD_Z = 5.05

  WALL_W = 12.0
  WALL_H = 8.0
  WALL_D = 0.25

  LETTER_SIZE = 0.72
  LETTER_DEPTH = 0.035

  LIGHT_COUNT = 6
  LIGHT_SPACING = 1.32

  CREATE_WALL = true
  CREATE_DOOR = true
  CREATE_OPTIONAL_OUTDOOR_SUPPORTS = false
  CREATE_LOGO = true

  # --------------------------------------------------------------
  # DON VI
  # --------------------------------------------------------------

  def m(value)
    value.to_f * 39.37007874015748
  end

  def pt(x, y, z)
    Geom::Point3d.new(m(x), m(y), m(z))
  end

  def vec(x, y, z)
    Geom::Vector3d.new(x, y, z)
  end

  def group(parent, name)
    g = parent.add_group
    g.name = name
    g
  end

  # --------------------------------------------------------------
  # VAT LIEU
  # --------------------------------------------------------------

  def material(name, rgb, alpha = 1.0)
    mats = Sketchup.active_model.materials
    mat = mats[name] || mats.add(name)
    mat.color = Sketchup::Color.new(*rgb)
    mat.alpha = alpha
    mat
  end

  def setup_materials
    @wall = material('SBM Dark Graphite Wall', [35, 46, 57])
    @wall_line = material('SBM Wall Seams', [62, 75, 86])

    @red = material('SBM Signboard Red', [205, 16, 28])
    @red_side = material('SBM Red Side Panels', [104, 9, 17])
    @red_detail = material('SBM Red Graphic Accent', [236, 35, 46])

    @white = material('SBM White Lettering', [255, 255, 252])
    @steel = material('SBM Dark Steel', [36, 44, 51])
    @aluminium = material('SBM Aluminium', [155, 166, 174])

    @lamp = material('SBM Lamp Housing', [24, 29, 33])
    @lamp_glow = material('SBM Lamp Glass', [255, 240, 190])

    @glass = material('SBM Glass', [80, 108, 122], 0.42)
    @frame = material('SBM Door Frame', [28, 38, 47])
    @concrete = material('SBM Concrete', [130, 132, 129])
    @logo = material('SBM Logo White', [255, 255, 255])
  end

  # --------------------------------------------------------------
  # KHOI HOP
  # --------------------------------------------------------------

  def box(entities, x, y, z, dx, dy, dz, mat)
    return nil if dx <= 0 || dy <= 0 || dz <= 0

    points = [
      pt(x, y, z),
      pt(x + dx, y, z),
      pt(x + dx, y + dy, z),
      pt(x, y + dy, z)
    ]

    face = entities.add_face(points)
    return nil unless face && face.valid?

    face.reverse! if face.normal.z < 0
    face.material = mat
    face.back_material = mat
    face.pushpull(m(dz))
    face
  end

  # --------------------------------------------------------------
  # THANH THEP NOI HAI DIEM
  # --------------------------------------------------------------

  def beam(entities, a, b, width, mat)
    direction = b - a
    length = direction.length
    return nil if length < 0.001

    direction.normalize!

    ref = vec(0, 0, 1)
    ref = vec(0, 1, 0) if direction.parallel?(ref)

    ux = direction.cross(ref)
    ux.normalize!
    uy = direction.cross(ux)
    uy.normalize!

    h = m(width) / 2.0

    corners = [
      a.offset(ux, h).offset(uy, h),
      a.offset(ux, -h).offset(uy, h),
      a.offset(ux, -h).offset(uy, -h),
      a.offset(ux, h).offset(uy, -h)
    ]

    face = entities.add_face(corners)
    return nil unless face && face.valid?

    face.material = mat
    face.back_material = mat
    face.pushpull(length)
    face
  end

  # --------------------------------------------------------------
  # TRU TRON
  # --------------------------------------------------------------

  def cylinder(entities, x, y, z, radius, height, mat, segments = 24)
    edges = entities.add_circle(
      pt(x, y, z),
      vec(0, 0, 1),
      m(radius),
      segments
    )

    face = entities.add_face(edges)
    return nil unless face && face.valid?

    face.material = mat
    face.back_material = mat
    face.pushpull(m(height))
    face
  end

  # --------------------------------------------------------------
  # TUONG MAT DUNG
  # --------------------------------------------------------------

  def build_wall(parent)
    g = group(parent, '01 - Dark Graphite Facade')
    e = g.entities

    box(
      e,
      -WALL_W / 2.0,
      0.12,
      0,
      WALL_W,
      WALL_D,
      WALL_H,
      @wall
    )

    # Duong ron ngang tuong
    15.times do |i|
      z = 0.35 + i * 0.48

      box(
        e,
        -WALL_W / 2.0,
        0.075,
        z,
        WALL_W,
        0.035,
        0.025,
        @wall_line
      )
    end

    # Duong ron dung ben trai
    box(
      e,
      -WALL_W / 2.0 + 0.28,
      0.065,
      0,
      0.045,
      0.035,
      WALL_H,
      @wall_line
    )

    g
  end

  # --------------------------------------------------------------
  # KHUNG XUONG BIEN HIEU
  # --------------------------------------------------------------

  def build_sign_frame(parent)
    g = group(parent, '02 - Signboard Steel Frame')
    e = g.entities

    x1 = BOARD_X - BOARD_W / 2.0
    x2 = BOARD_X + BOARD_W / 2.0
    y = BOARD_Y + BOARD_D
    z1 = BOARD_Z
    z2 = BOARD_Z + BOARD_H

    # Khung chu nhat
    beam(e, pt(x1, y, z1), pt(x2, y, z1), 0.07, @steel)
    beam(e, pt(x1, y, z2), pt(x2, y, z2), 0.07, @steel)
    beam(e, pt(x1, y, z1), pt(x1, y, z2), 0.07, @steel)
    beam(e, pt(x2, y, z1), pt(x2, y, z2), 0.07, @steel)

    # Thanh dung ben trong
    7.times do |i|
      x = x1 + (x2 - x1) * i / 6.0

      beam(
        e,
        pt(x, y, z1 + 0.08),
        pt(x, y, z2 - 0.08),
        0.035,
        @aluminium
      )
    end

    # Thanh ngang gia cuong
    [0.18, BOARD_H / 2.0, BOARD_H - 0.18].each do |dz|
      beam(
        e,
        pt(x1 + 0.06, y, z1 + dz),
        pt(x2 - 0.06, y, z1 + dz),
        0.035,
        @aluminium
      )
    end

    g
  end

  # --------------------------------------------------------------
  # THAN BIEN HIEU
  # --------------------------------------------------------------

  def build_sign_body(parent)
    g = group(parent, '03 - Red Signboard Housing')
    e = g.entities

    x = BOARD_X - BOARD_W / 2.0
    y = BOARD_Y
    z = BOARD_Z

    # Than hop bien do
    box(e, x, y, z, BOARD_W, BOARD_D, BOARD_H, @red)

    # Canh ben trai tao chieu sau
    box(
      e,
      x - 0.10,
      y + 0.015,
      z,
      0.12,
      BOARD_D + 0.04,
      BOARD_H,
      @red_side
    )

    # Vien tren va duoi
    box(
      e,
      x,
      y - 0.035,
      z + BOARD_H - 0.045,
      BOARD_W,
      0.045,
      0.045,
      @red_side
    )

    box(
      e,
      x,
      y - 0.035,
      z,
      BOARD_W,
      0.045,
      0.045,
      @red_side
    )

    # Duong cheo trang tri mat truoc
    front_y = y - 0.045

    beam(
      e,
      pt(BOARD_X + 2.15, front_y, z + BOARD_H - 0.08),
      pt(BOARD_X + 3.05, front_y, z + 0.08),
      0.10,
      @red_detail
    )

    beam(
      e,
      pt(BOARD_X + 2.50, front_y - 0.006, z + BOARD_H - 0.08),
      pt(BOARD_X + 3.40, front_y - 0.006, z + 0.08),
      0.045,
      @red_side
    )

    g
  end

  # --------------------------------------------------------------
  # CHU NOI 3D
  # --------------------------------------------------------------

  def add_text(parent, text, height, x, y, z, mat, font = 'Arial')
    g = group(parent, "Raised Text - #{text}")

    begin
      g.entities.add_3d_text(
        text,
        TextAlignLeft,
        font,
        false,
        true,
        m(height),
        0.0,
        0.0,
        true,
        m(LETTER_DEPTH)
      )

      g.entities.each do |entity|
        next unless entity.respond_to?(:material=)

        entity.material = mat
        entity.back_material = mat if entity.respond_to?(:back_material=)
      end

      # Chu mac dinh tao tren mat phang XY.
      # Xoay de mat chu huong ra phia truoc (truc -Y).
      rotate = Geom::Transformation.rotation(
        ORIGIN,
        X_AXIS,
        90.degrees
      )

      move = Geom::Transformation.translation(pt(x, y, z))
      g.transform!(move * rotate)
    rescue StandardError => error
      puts "Khong tao duoc text '#{text}': #{error.message}"
    end

    g
  end

  def build_sign_text(parent)
    g = group(parent, '04 - Raised White Sign Typography')
    x = BOARD_X - BOARD_W / 2.0 + 0.65
    y = BOARD_Y - 0.07
    z = BOARD_Z + 1.43

    add_text(
      g,
      'Sign Board',
      0.78,
      x,
      y,
      z,
      @white,
      'Arial'
    )

    add_text(
      g,
      'Mockup',
      0.78,
      x,
      y,
      z - 0.86,
      @white,
      'Arial'
    )

    # Logo hinh hoc don gian ben phai
    if CREATE_LOGO
      build_logo(g)
    end

    g
  end

  # --------------------------------------------------------------
  # LOGO TRANG TRI
  # --------------------------------------------------------------

  def build_logo(parent)
    g = group(parent, 'White Geometric Logo')
    e = g.entities

    x = BOARD_X + BOARD_W / 2.0 - 0.62
    y = BOARD_Y - 0.085
    z = BOARD_Z + 0.34

    # Hinh dang logo cach dieu tu cac thanh hop
    beam(
      e,
      pt(x - 0.20, y, z + 0.20),
      pt(x + 0.05, y, z + 0.43),
      0.11,
      @logo
    )

    beam(
      e,
      pt(x + 0.05, y, z + 0.43),
      pt(x + 0.22, y, z + 0.24),
      0.11,
      @logo
    )

    beam(
      e,
      pt(x + 0.22, y, z + 0.24),
      pt(x + 0.03, y, z + 0.03),
      0.11,
      @logo
    )

    beam(
      e,
      pt(x + 0.03, y, z + 0.03),
      pt(x - 0.14, y, z + 0.16),
      0.11,
      @logo
    )

    g
  end

  # --------------------------------------------------------------
  # DEN CHIEU SANG
  # --------------------------------------------------------------

  def build_lighting(parent)
    g = group(parent, '05 - Signboard Spotlights')
    e = g.entities

    count = LIGHT_COUNT
    first_x = BOARD_X - (count - 1) * LIGHT_SPACING / 2.0

    count.times do |i|
      x = first_x + i * LIGHT_SPACING
      z = BOARD_Z + BOARD_H + 0.28

      # Chan gan den vao tuong
      cylinder(
        e,
        x,
        0.015,
        z + 0.10,
        0.065,
        0.07,
        @steel,
        16
      )

      # Tay den huong ra phia truoc
      beam(
        e,
        pt(x, 0.05, z + 0.10),
        pt(x, -0.42, z + 0.02),
        0.035,
        @lamp
      )

      # Than den
      box(
        e,
        x - 0.13,
        -0.50,
        z - 0.05,
        0.26,
        0.16,
        0.09,
        @lamp
      )

      # Mat phat sang
      box(
        e,
        x - 0.105,
        -0.515,
        z - 0.067,
        0.21,
        0.13,
        0.025,
        @lamp_glow
      )
    end

    g
  end

  # --------------------------------------------------------------
  # CUA KINH PHIA DUOI
  # --------------------------------------------------------------

  def build_door(parent)
    g = group(parent, '06 - Glass Entrance and Frames')
    e = g.entities

    door_x = -3.1
    door_z = 0.0
    door_w = 6.2
    door_h = 3.1
    y = -0.005

    # Khung bao ngoai
    box(
      e,
      door_x,
      y,
      door_z,
      door_w,
      0.10,
      0.12,
      @frame
    )

    box(
      e,
      door_x,
      y,
      door_z,
      0.12,
      0.10,
      door_h,
      @frame
    )

    box(
      e,
      door_x + door_w - 0.12,
      y,
      door_z,
      0.12,
      0.10,
      door_h,
      @frame
    )

    box(
      e,
      door_x,
      y,
      door_h - 0.12,
      door_w,
      0.10,
      0.12,
      @frame
    )

    # Hai canh cua kinh
    pane_w = (door_w - 0.28) / 2.0

    2.times do |i|
      px = door_x + 0.14 + i * (pane_w + 0.02)

      box(
        e,
        px,
        -0.015,
        door_z + 0.14,
        pane_w,
        0.025,
        door_h - 0.28,
        @glass
      )

      # Thanh dung giua moi canh
      box(
        e,
        px,
        -0.06,
        door_z + 0.14,
        0.035,
        0.04,
        door_h - 0.28,
        @frame
      )
    end

    g
  end

  # --------------------------------------------------------------
  # KHUNG CHAN BIEN / GIA DO
  # --------------------------------------------------------------

  def build_support_brackets(parent)
    g = group(parent, '07 - Signboard Mounting Brackets')
    e = g.entities

    x1 = BOARD_X - BOARD_W / 2.0 + 0.50
    x2 = BOARD_X + BOARD_W / 2.0 - 0.50

    [x1, x2].each do |x|
      # Tay do tren
      beam(
        e,
        pt(x, 0.15, BOARD_Z + 0.35),
        pt(x, -0.38, BOARD_Z + 0.15),
        0.08,
        @steel
      )

      # Tay do duoi
      beam(
        e,
        pt(x, 0.15, BOARD_Z + 0.10),
        pt(x, -0.38, BOARD_Z + 0.15),
        0.08,
        @steel
      )

      # Ban ma gan tuong
      box(
        e,
        x - 0.14,
        0.025,
        BOARD_Z - 0.05,
        0.28,
        0.10,
        0.60,
        @aluminium
      )

      # Bu long
      [BOARD_Z + 0.06, BOARD_Z + 0.42].each do |z|
        cylinder(
          e,
          x,
          -0.035,
          z,
          0.035,
          0.045,
          @steel,
          12
        )
      end
    end

    g
  end

  # --------------------------------------------------------------
  # SAN CATWALK PHU
  # --------------------------------------------------------------

  def build_catwalk(parent)
    g = group(parent, '08 - Maintenance Catwalk')
    e = g.entities

    width = BOARD_W + 0.35
    depth = 0.70
    x = BOARD_X - width / 2.0
    y = BOARD_Y + BOARD_D + 0.05
    z = BOARD_Z - 0.34

    # San thao tac
    box(
      e,
      x,
      y,
      z,
      width,
      depth,
      0.08,
      @steel_dark
    )

    # Dam doc
    [y, y + depth].each do |dy|
      beam(
        e,
        pt(x, dy, z - 0.02),
        pt(x + width, dy, z - 0.02),
        0.07,
        @steel
      )
    end

    # Dam ngang
    13.times do |i|
      px = x + width * i / 12.0

      beam(
        e,
        pt(px, y, z - 0.02),
        pt(px, y + depth, z - 0.02),
        0.035,
        @steel
      )
    end

    # Lan can phia ngoai
    rail_z = z + 0.95
    outer_y = y + depth

    beam(
      e,
      pt(x, outer_y, rail_z),
      pt(x + width, outer_y, rail_z),
      0.04,
      @aluminium
    )

    beam(
      e,
      pt(x, outer_y, z + 0.47),
      pt(x + width, outer_y, z + 0.47),
      0.03,
      @steel
    )

    13.times do |i|
      px = x + width * i / 12.0

      beam(
        e,
        pt(px, outer_y, z + 0.05),
        pt(px, outer_y, rail_z),
        0.025,
        @steel
      )
    end

    g
  end

  # --------------------------------------------------------------
  # THANG LEO PHIA SAU
  # --------------------------------------------------------------

  def build_ladder(parent)
    g = group(parent, '09 - Rear Access Ladder')
    e = g.entities

    x = BOARD_X + BOARD_W / 2.0 + 0.30
    y = BOARD_Y + BOARD_D + 0.35
    z0 = 0.30
    z1 = BOARD_Z + BOARD_H - 0.10

    [-0.18, 0.18].each do |dx|
      beam(
        e,
        pt(x + dx, y, z0),
        pt(x + dx, y, z1),
        0.035,
        @steel
      )
    end

    z = z0

    while z <= z1
      beam(
        e,
        pt(x - 0.18, y, z),
        pt(x + 0.18, y, z),
        0.025,
        @aluminium
      )

      z += 0.30
    end

    g
  end

  # --------------------------------------------------------------
  # MONG VA COT NGOAI TROI TUY CHON
  # --------------------------------------------------------------

  def build_optional_supports(parent)
    g = group(parent, '10 - Optional Outdoor Foundations and Columns')
    e = g.entities

    [-3.0, 3.0].each do |x|
      # Mong
      box(
        e,
        x - 0.65,
        -0.75,
        -1.0,
        1.30,
        1.30,
        1.0,
        @concrete
      )

      # Co mong
      box(
        e,
        x - 0.45,
        -0.45,
        0,
        0.90,
        0.90,
        0.30,
        @concrete
      )

      # Ban de
      box(
        e,
        x - 0.38,
        -0.38,
        0.30,
        0.76,
        0.76,
        0.07,
        @steel
      )

      # Cot tron
      cylinder(
        e,
        x,
        0.0,
        0.37,
        0.28,
        BOARD_Z - 0.25,
        @aluminium,
        36
      )

      # Ban ma dau cot
      box(
        e,
        x - 0.40,
        -0.40,
        BOARD_Z - 0.10,
        0.80,
        0.80,
        0.12,
        @steel_dark
      )

      # Bu long chan cot
      [-0.25, 0.25].each do |dx|
        [-0.25, 0.25].each do |dy|
          cylinder(
            e,
            x + dx,
            dy,
            0.30,
            0.025,
            0.22,
            @steel,
            12
          )
        end
      end
    end

    g
  end

  # --------------------------------------------------------------
  # CAMERA
  # --------------------------------------------------------------

  def set_camera
    model = Sketchup.active_model
    view = model.active_view

    eye = pt(10.0, -18.0, 8.0)
    target = pt(0.0, 0.0, 4.6)
    up = vec(0, 0, 1)

    camera = Sketchup::Camera.new(eye, target, up)
    view.camera = camera
    view.perspective = true
    view.zoom_extents
    view.invalidate
  end

  # --------------------------------------------------------------
  # TAO MO HINH
  # --------------------------------------------------------------

  def build
    model = Sketchup.active_model
    model.start_operation('Create Signboard Mockup', true)

    begin
      setup_materials

      root = group(
        model.active_entities,
        'SIGNBOARD MOCKUP - COMPLETE MODEL'
      )

      build_wall(root) if CREATE_WALL
      build_sign_frame(root)
      build_sign_body(root)
      build_sign_text(root)
      build_lighting(root)
      build_door(root) if CREATE_DOOR
      build_support_brackets(root)
      build_catwalk(root)
      build_ladder(root)

      if CREATE_OPTIONAL_OUTDOOR_SUPPORTS
        build_optional_supports(root)
      end

      model.commit_operation
      set_camera

      puts '============================================='
      puts 'SIGNBOARD MOCKUP CREATED'
      puts "Bien hieu: #{BOARD_W}m x #{BOARD_H}m"
      puts "Do day than bien: #{BOARD_D}m"
      puts "So den chieu sang: #{LIGHT_COUNT}"
      puts 'Mo hinh da duoc phan nhom theo tung bo phan.'
      puts '============================================='

    rescue StandardError => error
      model.abort_operation
      puts "SIGNBOARD ERROR: #{error.message}"
      puts error.backtrace
    end
  end
end
